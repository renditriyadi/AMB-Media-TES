Gunakan script otomatis di bawah ini untuk mengekstrak Refresh Token & Cookie akun Weavy kamu secara instan tanpa perlu mencari manual di DevTools:

  1. Buka tab https://app.weavy.ai di browser dan pastikan sudah login.
  2. Buka Developer Tools (Tekan F12 atau Ctrl+Shift+I) → pilih tab Console.
  3. Klik tombol "Salin Script" di bawah, paste ke Console Weavy, lalu tekan Enter.
  4. Token & Cookie akan otomatis disalin ke clipboard! Kembali ke sini, klik + Add Account lalu tempelkan.
Console Script (Auto Extract Token) 
(async function(){
  try {
    if (!location.host.includes("weavy.ai")) {
      alert("⚠️ Buka app.weavy.ai dulu di tab ini!");
      return;
    }
    var getReq = function(req) {
      return new Promise(function(res, rej) {
        req.onsuccess = function() { res(req.result); };
        req.onerror = function() { rej(req.error); };
      });
    };
    var db = await getReq(indexedDB.open("firebaseLocalStorageDb"));
    var store = db.transaction("firebaseLocalStorage", "readonly").objectStore("firebaseLocalStorage");
    var items = await getReq(store.getAll());
    db.close();
    var target = null;
    for (var i = 0; i < items.length; i++) {
      var val = (items[i] && items[i].value) ? items[i].value : items[i];
      if (val && val.stsTokenManager && val.stsTokenManager.refreshToken) {
        target = val;
        break;
      }
    }
    if (!target) {
      alert("⚠️ Refresh Token tidak ditemukan. Pastikan Anda sudah login di app.weavy.ai");
      return;
    }
    var payload = {
      refreshToken: target.stsTokenManager.refreshToken,
      email: target.email || "",
      cookie: document.cookie || ""
    };
    var str = JSON.stringify(payload);
    try {
      await navigator.clipboard.writeText(str);
      alert("✅ BERHASIL DISALIN!\n\nData Token & Cookie akun " + (payload.email || "Weavy") + " sudah disalin ke Clipboard.\n\nSilakan buka Ekstensi lalu tempelkan ke kolom Refresh Token!");
    } catch(errCopy) {
      prompt("Salin data JSON berikut:", str);
    }
  } catch(e) {
    alert("❌ Error: " + e.message);
  }
})();
